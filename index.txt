        function renderGatekeepers() {
            const container = document.getElementById('gatekeeper-list');
            container.innerHTML = '';
            gatekeeperData.forEach((g) => {
                const isChecked = !!state.gatekeeper[g.id];
                const div = document.createElement('div');
                div.className = `p-4 hover:bg-slate-50 transition-colors flex items-start gap-3 cursor-pointer ${isChecked ? 'bg-indigo-50/40' : ''}`;
                div.onclick = (e) => {
                    if (e.target.tagName !== 'INPUT') {
                        const chk = document.getElementById(`chk-${g.id}`);
                        chk.checked = !chk.checked;
                        toggleGatekeeper(g.id, chk.checked);
                    }
                };
                div.innerHTML = `
                    <input type="checkbox" id="chk-${g.id}" ${isChecked ? 'checked' : ''} 
                        onchange="toggleGatekeeper('${g.id}', this.checked)"
                        class="mt-1 w-5 h-5 rounded border-slate-300 text-indigo-600 focus:ring-indigo-500 cursor-pointer shrink-0">
                    <label for="chk-${g.id}" class="text-sm font-medium text-slate-700 cursor-pointer leading-relaxed select-none flex-grow">

                    </label>
                `;

            });
        }

        function toggleGatekeeper(id, checked) {

            saveState();

        }

        function handleItemClick(e, sectionKey, critId, itemIdx) {
            if (e.target.tagName !== 'INPUT') {
                const chk = document.getElementById(`chk-${sectionKey}-${critId}-${itemIdx}`);
                chk.checked = !chk.checked;
                toggleItem(sectionKey, critId, itemIdx, chk.checked);
            }
        }

        function toggleItem(sectionKey, critId, itemIdx, checked) {
            if (!state[sectionKey][critId]) {
                state[sectionKey][critId] = [];
            }
            if (checked) {
                if (!state[sectionKey][critId].includes(itemIdx)) {
                    state[sectionKey][critId].push(itemIdx);
                }
            } else {
                state[sectionKey][critId] = state[sectionKey][critId].filter(i => i !== itemIdx);
            }
            saveState();
            calculateScores();
            
            // Update individual headers without re-rendering everything
            const crit = [...reportCriteria, ...exhibitionCriteria].find(x => x.id === critId);
            if (crit) {
                const checkedCount = state[sectionKey][critId].length;
                const earnedPoints = checkedCount * crit.weight;
                document.getElementById(`count-${critId}`).innerText = checkedCount;
                document.getElementById(`score-${critId}`).innerText = earnedPoints;
            }
        }

        function calculateScores() {

            const passGatekeeperCount = Object.values(state.gatekeeper).filter(Boolean).length;

            document.getElementById('gatekeeper-count').innerText = passGatekeeperCount;

            // 2. Report Score
            let reportTotal = 0;
            reportCriteria.forEach(c => {
                const count = (state.report[c.id] || []).length;
                reportTotal += count * c.weight;
            });

            // 3. Exhibition Score
            let exhibitionTotal = 0;
            exhibitionCriteria.forEach(c => {
                const count = (state.exhibition[c.id] || []).length;
                exhibitionTotal += count * c.weight;

            const grandTotal = reportTotal + exhibitionTotal;

            document.getElementById('report-score-total').innerText = reportTotal;
            document.getElementById('exhibition-score-total').innerText = exhibitionTotal;

            // Gatekeeper Banner Status

            } else if (grandTotal >= 50) {

        function generateRecommendations() {
            const container = document.getElementById('recommendations-list');
            container.innerHTML = '';

            let recs = [];

            // Check Report Unchecked Items
            reportCriteria.forEach(c => {
                const checkedIndices = state.report[c.id] || [];
                if (checkedIndices.length < 5) {
                    const missingCount = 5 - checkedIndices.length;
                    recs.push({
                        category: "📄 เล่มรายงาน",
                        title: c.title,
                        missingCount: missingCount,
                        tip: c.tip,
                        weight: c.weight
                    });
                }
            });

            // Check Exhibition Unchecked Items
            exhibitionCriteria.forEach(c => {
                const checkedIndices = state.exhibition[c.id] || [];
                if (checkedIndices.length < 5) {
                    const missingCount = 5 - checkedIndices.length;
                    recs.push({
                        category: "🎪 นิทรรศการและการนำเสนอ",
                        title: c.title,
                        missingCount: missingCount,
                        tip: c.tip,
                        weight: c.weight
                    });
                }
            });

            if (recs.length === 0) {
                container.innerHTML = `
                    <div class="p-6 text-center bg-emerald-50 rounded-xl border border-emerald-200 text-emerald-800 space-y-2">
                        <i data-lucide="award" class="w-10 h-10 text-emerald-600 mx-auto"></i>
                        <h4 class="font-bold font-heading text-lg">ยอดเยี่ยม! ผลงานของคุณมีองค์ประกอบครบถ้วน 100%</h4>
                        <p class="text-xs text-emerald-700">ผลงานมีโอกาสได้รับรางวัลเหรียญทองชนะเลิศ ขอให้เตรียมตัวนำเสนอหน้างานอย่างมั่นใจ</p>
                    </div>
                `;
                return;
            }

            // Sort recommendations by impact (missing points potential)
            recs.sort((a, b) => (b.missingCount * b.weight) - (a.missingCount * a.weight));

            recs.forEach((rec) => {
                const potentialLostPoints = rec.missingCount * rec.weight;
                const div = document.createElement('div');
                div.className = "p-4 rounded-xl border border-slate-200 bg-slate-50/50 hover:bg-slate-50 transition-all flex flex-col sm:flex-row sm:items-center justify-between gap-3";
                div.innerHTML = `
                    <div class="space-y-1">
                        <div class="flex items-center gap-2">
                            <span class="text-[11px] font-semibold px-2 py-0.5 rounded bg-slate-200 text-slate-700">${rec.category}</span>
                            <h4 class="font-bold text-slate-800 text-sm font-heading">${rec.title}</h4>
                        </div>
                        <p class="text-xs text-slate-600 leading-relaxed"><strong class="text-amber-700">คำแนะนำ:</strong> ${rec.tip}</p>
                    </div>
                    <div class="shrink-0 self-end sm:self-auto text-right">
                        <span class="text-xs font-medium text-rose-600 bg-rose-50 border border-rose-200 px-2.5 py-1 rounded-lg inline-block">
                            ขาดอีก ${rec.missingCount} ข้อ (-${potentialLostPoints} คะแนน)
                        </span>
                    </div>
                `;
                container.appendChild(div);
            });
        }

        function updateChart() {
            if (!chartInstance) return;

            const scores = [];

            // Report 8 Criteria
            reportCriteria.forEach(c => {
                const count = (state.report[c.id] || []).length;
                scores.push(count * c.weight);
            });

            // Exhibition 4 Criteria
            exhibitionCriteria.forEach(c => {
                const count = (state.exhibition[c.id] || []).length;
                scores.push(count * c.weight);
            });

            chartInstance.data.datasets[0].data = scores;
            chartInstance.update();
        }

        function saveState() {
            try {
                localStorage.setItem('cm1_innovation_assessment_2026', JSON.stringify(state));
            } catch (e) {
                console.error("Local storage save error", e);
            }
        }

        function loadState() {
            try {
                const saved = localStorage.getItem('cm1_innovation_assessment_2026');
                if (saved) {
                    state = JSON.parse(saved);
                } else {
                    // Default initial empty state
                    state = { gatekeeper: {}, report: {}, exhibition: {} };
                }
            } catch (e) {
                state = { gatekeeper: {}, report: {}, exhibition: {} };
            }
        }

        function resetForm() {
            if (confirm("คุณต้องการล้างข้อมูลการประเมินตนเองทั้งหมดใช่หรือไม่?")) {
                state = { gatekeeper: {}, report: {}, exhibition: {} };
                saveState();
                renderGatekeepers();
                renderCriteria('report-criteria-container', reportCriteria, 'report');
                renderCriteria('exhibition-criteria-container', exhibitionCriteria, 'exhibition');
                calculateScores();
                updateChart();
                switchTab('gatekeeper');
            }
        }
